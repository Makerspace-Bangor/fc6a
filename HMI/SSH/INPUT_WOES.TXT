#!/bin/sh

GAME=/home/service/v/games/vitetris
TOUCH=/dev/input/event1

export XDG_RUNTIME_DIR=/tmp/0-runtime-dir
export WAYLAND_DISPLAY=wayland-0

# Launch Vitetris in Weston
weston-terminal --shell="$GAME" &
WESTON_PID=$!

# Wait for Vitetris to start
PID=""
for i in 1 2 3 4 5; do
    PID=$(pidof vitetris | awk '{print $1}')
    [ -n "$PID" ] && break
    sleep 1
done

if [ -z "$PID" ]; then
    echo "Vitetris failed to start"
    exit 1
fi

echo "Vitetris PID: $PID"

# Find its terminal
TTY=$(readlink /proc/$PID/fd/0)
MASTER="/dev/pty${TTY##*/tty}"

echo "Terminal: $TTY"
echo "Master:   $MASTER"

if [ ! -e "$MASTER" ]; then
    echo "PTY master not found"
    exit 1
fi

echo "Touch the screen to send ENTER"

# Decode touchscreen events
hexdump -v -e '16/1 "%02x " "\n"' "$TOUCH" |
awk '
$9=="03" && $10=="00" && $11=="35" && $12=="00" {
    touching=1
    if (!down) {
        down=1
        print "ENTER"
        fflush()
    }
}
$9=="00" && $10=="00" && $11=="00" && $12=="00" {
    if (!touching)
        down=0
    touching=0
}' |
while read -r KEY; do
    echo "Touch detected: $KEY"

    if ! printf '\r' > "$MASTER"; then
        echo "Cannot write to $MASTER"
        echo "Weston may have exclusive ownership"
        break
    fi
done
