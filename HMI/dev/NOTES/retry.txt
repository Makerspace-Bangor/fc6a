# Find all the dlls with .net assemblies, and decompile 
mkdir ~/ao51-decompiled

find ~/AO -type f \( -iname '*.dll' -o -iname '*.exe' \) -print0 |
while IFS= read -r -d '' f; do
    if file "$f" | grep -qiE 'Mono/\.Net assembly|\.Net assembly'; then
        rel="${f#$HOME/AO/}"
        out="$HOME/ao51-decompiled/${rel%.*}.il"

        mkdir -p "$(dirname "$out")"

        echo "Decompiling: $rel"

        if ! monodis "$f" > "$out" 2>"$out.err"; then
            echo "  FAILED: $rel"
        else
            rm -f "$out.err"
        fi
    fi
done
