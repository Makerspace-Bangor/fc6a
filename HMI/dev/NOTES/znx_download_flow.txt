		private ResultCode DownloadDetailDataForLinux(PDUserDataInfo userdataInfo, ChangeProjectSettingInfo changeInfo = null, Action<PDProcessingStage, ResultCode, int, string, int, int> progress = null, bool isExecuteLadder = false)
		{
			ResultCode resultCode = ResultCode.Success;
			try
			{
				_ReportProgress(progress, PDProcessingStage.ChangingSystemMode, ResultCode.Continue, 40);
				resultCode = ChangingMode(PDMode.DataTransmissionMode);
				if (resultCode != ResultCode.Success)
				{
					_ReportProgress(progress, PDProcessingStage.ChangingSystemMode, ResultCode.Failure);
					return resultCode;
				}
				_ReportProgress(progress, PDProcessingStage.ChangingSystemMode, ResultCode.Success, 45);
				resultCode = PDProtocol.Initialization(PDDataType.SelectableInitialization, userdataInfo.DataNo);
				if (resultCode != ResultCode.Success)
				{
					return resultCode;
				}
				_ReportProgress(progress, PDProcessingStage.ConnectingToOperatorInterface, ResultCode.Continue, 50);
				resultCode = InitialFTPConnection();
				if (resultCode != ResultCode.Success)
				{
					_ReportProgress(progress, PDProcessingStage.ConnectingToOperatorInterface, ResultCode.Failure);
					return resultCode;
				}
				_ReportProgress(progress, PDProcessingStage.ConnectingToOperatorInterface, ResultCode.Success, 55);
				if (resultCode == ResultCode.Success)
				{
					StartLFThread();
					_ReportProgress(progress, PDProcessingStage.TransferringTheData, ResultCode.Continue, 60);
					resultCode = SendProjectDataForLinux();
					if (resultCode == ResultCode.Success)
					{
						_ReportProgress(progress, PDProcessingStage.TransferringTheData, ResultCode.Success, 65);
					}
					else
					{
						_ReportProgress(progress, PDProcessingStage.TransferringTheData, ResultCode.Failure, 65);
					}
				}
				StopLFThread();
				if (resultCode == ResultCode.Success)
				{
					_ReportProgress(progress, PDProcessingStage.CheckingTheSendAndReceiveData, ResultCode.Continue, 70);
					resultCode = PDProtocol.SendLCCommand(LinuxDownloadMode.Normal);
					if (resultCode != ResultCode.Success)
					{
						_ReportProgress(progress, PDProcessingStage.CheckingTheSendAndReceiveData, ResultCode.Failure);
					}
					_ReportProgress(progress, PDProcessingStage.CheckingTheSendAndReceiveData, ResultCode.Success, 75);
				}
				_ReportProgress(progress, PDProcessingStage.DisconnectingTheConnection, ResultCode.Continue, 80);
				SendLDIfNecessary(resultCode == ResultCode.Success);
				_ReportProgress(progress, PDProcessingStage.DisconnectingTheConnection, ResultCode.Success, 85);
				if (changeInfo.IPSettingType == "Do not change")
				{
					_ReportProgress(progress, PDProcessingStage.ChangeProjectSetting);
					resultCode = PDProtocol.ChangeProjectSetting(PDOILinkType.None, PDOILinkSpeed.None, changeInfo.IpAddress, changeInfo.SubnetMask, changeInfo.DefaultGateway, changeInfo.PortNo, oldVer: false, changeInfo.IsChangeIP, changeInfo.IsChangeSubnetMask, changeInfo.IsChangeDefaultGateway, changeInfo.IsChangePotrtNo);
					if (resultCode != ResultCode.Success)
					{
						_ReportProgress(progress, PDProcessingStage.ChangeProjectSetting, ResultCode.Failure);
						return resultCode;
					}
					_ReportProgress(progress, PDProcessingStage.ChangeProjectSetting, ResultCode.Success, 90);
				}
				if (Helper.IsInnerPLCSeries)
				{
					if (isExecuteLadder)
					{
						_ReportProgress(progress, PDProcessingStage.ExcuteLadderProgram, ResultCode.Continue, 95);
					}
					resultCode = SendLGCommand(isExecuteLadder);
					if (isExecuteLadder)
					{
						if (resultCode == ResultCode.Success)
						{
							_ReportProgress(progress, PDProcessingStage.ExcuteLadderProgram, ResultCode.Success, 96);
						}
						else
						{
							_ReportProgress(progress, PDProcessingStage.ExcuteLadderProgram, ResultCode.Failure);
						}
					}
				}
				_ReportProgress(progress, PDProcessingStage.BackToRunningMode);
				resultCode = PDProtocol.BackToRunningMode();
				if (resultCode != ResultCode.Success)
				{
					_ReportProgress(progress, PDProcessingStage.BackToRunningMode, ResultCode.Failure);
				}
			}
			catch
			{
				resultCode = ResultCode.Failure;
			}
			finally
			{
				SendStopLFAndLDCommand();
			}
			return resultCode;
		}

				if (currentOS == 0)
				{
					_ReportProgress(progress, PDProcessingStage.ChangingSystemMode, ResultCode.Continue, 40);
					resultCode = ChangingMode(PDMode.SystemTransmissionMode);
					if (resultCode != ResultCode.Success)
					{
						_ReportProgress(progress, PDProcessingStage.ChangingSystemMode, ResultCode.Failure);
						return resultCode;
					}
					_ReportProgress(progress, PDProcessingStage.ChangingSystemMode, ResultCode.Success, 45);
				}
				_ReportProgress(progress, PDProcessingStage.ErasingTheData, ResultCode.Continue, 50);
				resultCode = PDProtocol.SendLECommand();
				if (resultCode != ResultCode.Success)
				{
					_ReportProgress(progress, PDProcessingStage.ErasingTheData, ResultCode.Failure);
					return resultCode;
				}
				_ReportProgress(progress, PDProcessingStage.ErasingTheData, ResultCode.Success, 55);
				if (resultCode == ResultCode.Success)
				{
					_ReportProgress(progress, PDProcessingStage.ConnectingToOperatorInterface, ResultCode.Continue, 60);
					resultCode = InitialFTPConnection();
					if (resultCode != ResultCode.Success)
					{
						_ReportProgress(progress, PDProcessingStage.ConnectingToOperatorInterface, ResultCode.Failure);
						return resultCode;
					}
					_ReportProgress(progress, PDProcessingStage.ConnectingToOperatorInterface, ResultCode.Success, 65);
					if (resultCode == ResultCode.Success)
					{
						StartLFThread();
						_ReportProgress(progress, PDProcessingStage.TransferringTheData, ResultCode.Continue, 70);
						Action<string, int, int, ResultCode> progressFile = (string fileName, int fileSize, int transferredSize, ResultCode ret) =>
						{
							_ReportProgress(progress, PDProcessingStage.TransferringTheData, ret, -1, fileName, fileSize, transferredSize);
						};
						resultCode = SendOSData(progressFile);
						if (resultCode == ResultCode.Success)
						{
							_ReportProgress(progress, PDProcessingStage.TransferringTheData, ResultCode.Success, 75);
						}
						else
						{
							_ReportProgress(progress, PDProcessingStage.TransferringTheData, ResultCode.Failure, 75);
						}
					}
					StopLFThread();
					if (resultCode == ResultCode.Success)
					{
						_ReportProgress(progress, PDProcessingStage.CheckingTheSendAndReceiveData, ResultCode.Continue, 80);
						resultCode = PDProtocol.SendLCCommand(LinuxDownloadMode.Kernel);
						if (resultCode != ResultCode.Success)
						{
							_ReportProgress(progress, PDProcessingStage.CheckingTheSendAndReceiveData, ResultCode.Failure);
						}
						_ReportProgress(progress, PDProcessingStage.CheckingTheSendAndReceiveData, ResultCode.Success, 85);
					}
					_ReportProgress(progress, PDProcessingStage.DisconnectingTheConnection, ResultCode.Continue, 90);
					SendLDIfNecessary(resultCode == ResultCode.Success);
					_ReportProgress(progress, PDProcessingStage.DisconnectingTheConnection, ResultCode.Success, 95);
					_ReportProgress(progress, PDProcessingStage.BackToRunningMode);
					resultCode = PDProtocol.BackToRunningMode();
					if (resultCode != ResultCode.Success)
					{
						_ReportProgress(progress, PDProcessingStage.BackToRunningMode, ResultCode.Failure);
					}
				}
			}
			catch
			{
				resultCode = ResultCode.Failure;
			}
			finally
			{
				SendStopLFAndLDCommand();
			}
			return resultCode;
		}
