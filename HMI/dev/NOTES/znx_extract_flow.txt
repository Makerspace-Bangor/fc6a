			return ResultCode.Success;
		}

		private void ReadZNXHeader(ZNXFile znxFile, BinaryReader br)
		{
			byte[] array = new byte[4];
			array = br.ReadBytes(array.Length);
			znxFile.FileType = Encoding.Default.GetString(array).TrimEnd(new char[1]);
			if ("ZNX" == znxFile.FileType)
			{
				znxFile.FormatVersion = br.ReadUInt16();
				znxFile.ReservedHeader = br.ReadUInt16();
				znxFile.OsVersion = GetOSVersion(br);
				znxFile.SystemSoftwareVersion = GetSystemSoftwareVersion(br);
				znxFile.SizeOfZNXFile = br.ReadUInt32();
				ZNXFileInformation zNXFileInformation = new ZNXFileInformation();
				zNXFileInformation.FileCotent = br.ReadUInt16();
				zNXFileInformation.ReservedFileInformation = br.ReadUInt16();
				zNXFileInformation.JumpSizeToTheFile = br.ReadUInt32();
				zNXFileInformation.FileSizeOfTheFile = br.ReadUInt32();
				zNXFileInformation.CheckSum = br.ReadUInt32();
				zNXFileInformation.SizeOfFileName = br.ReadUInt32();
				byte[] array2 = new byte[zNXFileInformation.SizeOfFileName];
				array2 = br.ReadBytes(array2.Length);
				zNXFileInformation.FileName = Encoding.Default.GetString(array2).TrimEnd(new char[1]);
				znxFile.FileInformation.Add(zNXFileInformation);
				ZNXFileInformation zNXFileInformation2 = new ZNXFileInformation();
				zNXFileInformation2.FileCotent = br.ReadUInt16();
				zNXFileInformation2.ReservedFileInformation = br.ReadUInt16();
				zNXFileInformation2.JumpSizeToTheFile = br.ReadUInt32();
				zNXFileInformation2.FileSizeOfTheFile = br.ReadUInt32();
				zNXFileInformation2.CheckSum = br.ReadUInt32();
				zNXFileInformation2.SizeOfFileName = br.ReadUInt32();
				array2 = new byte[zNXFileInformation2.SizeOfFileName];
				array2 = br.ReadBytes(array2.Length);
				zNXFileInformation2.FileName = Encoding.Default.GetString(array2).TrimEnd(new char[1]);
				znxFile.FileInformation.Add(zNXFileInformation2);
				br.BaseStream.Seek(zNXFileInformation.FileSizeOfTheFile, SeekOrigin.Current);
			}
		}

		private string GetOSVersion(BinaryReader br)
		{
			int num = br.ReadInt16();
			int num2 = br.ReadInt16();
			int num3 = br.ReadInt16();
			int num4 = br.ReadInt16();
			return string.Format("{0}.{1}.{2}.{3}", new object[4] { num, num2, num3, num4 });
		}

		private string GetSystemSoftwareVersion(BinaryReader br)
		{
			int num = br.ReadInt16();
			int num2 = br.ReadInt16();
			int num3 = br.ReadInt16();
			int num4 = br.ReadInt16();
			return string.Format("{0}.{1}.{2}.{3}", new object[4] { num, num2, num3, num4 });
		}

		private ResultCode CalcProjectDataSize(string[] filePaths, out uint size)
		{
			size = 0u;
			try
			{
				foreach (string text in filePaths)
				{
					switch (Path.GetExtension(text))
					{
					case ".b2f":
						size += (uint)Math.Ceiling((decimal)new FileInfo(text).Length / 131072m) * 131072;
						break;
					default:
						size += (uint)((int)new FileInfo(text).Length + 8);
